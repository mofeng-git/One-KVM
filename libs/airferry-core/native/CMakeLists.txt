cmake_minimum_required(VERSION 3.22)
project(one_kvm_airferry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZXING_READERS ON CACHE BOOL "" FORCE)
set(ZXING_WRITERS OFF CACHE STRING "" FORCE)
set(ZXING_C_API OFF CACHE BOOL "" FORCE)
foreach(format 1D AZTEC DATAMATRIX MAXICODE PDF417)
  set(ZXING_ENABLE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(vendor/zxing)
add_library(airferry_native STATIC jpeg_qr.cpp vendor/airferry/airferry_zxing_core.cpp)
target_include_directories(airferry_native PRIVATE vendor/airferry ${TURBOJPEG_INCLUDE_DIRS})
target_link_libraries(airferry_native PRIVATE ZXing::ZXing)
if(AIRFERRY_FIXTURES)
  target_compile_definitions(airferry_native PRIVATE AIRFERRY_FIXTURES)
endif()
if(MSVC)
  target_compile_options(airferry_native PRIVATE /O2 /EHsc)
else()
  target_compile_options(airferry_native PRIVATE -O3 -fexceptions)
endif()
install(TARGETS airferry_native ZXing ARCHIVE DESTINATION lib)
